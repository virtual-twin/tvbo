import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    omega = _p.omega
    f = 500 * omega / jnp.pi

    # Coupling
    c_glob = cX[0]
    local_coupling = 0

    # State variables
    theta = current_state[0]

    # Derived variables
    lc_0 = jnp.sin(local_coupling * theta)
    phase = (1 / 2) * theta / jnp.pi
    signal = jnp.sin(theta)
    I = c_glob + lc_0

    # Derivatives
    dtheta_dt = I + omega

    derivatives = jnp.array(jnp.broadcast_arrays(dtheta_dt))

    return derivatives
