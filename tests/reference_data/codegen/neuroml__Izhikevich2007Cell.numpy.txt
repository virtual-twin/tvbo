import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    C = _p.C
    k = _p.k
    vr = _p.vr
    vt = _p.vt
    a = _p.a
    b = _p.b

    # State variables
    v = current_state[0]
    u = current_state[1]

    # Derived variables
    I_ext = np.where(
        np.logical_and.reduce((np.greater_equal(t, 0.05), np.less(t, 0.25))),
        1000.0,
        0.0,
    )

    # Derivatives
    dv_dt = (I_ext - u + k * (v - vr) * (v - vt)) / C
    du_dt = a * (-u + b * (v - vr))

    derivatives = np.array(np.broadcast_arrays(dv_dt, du_dt))

    return derivatives
