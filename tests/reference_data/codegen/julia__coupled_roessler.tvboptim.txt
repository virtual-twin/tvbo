import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class CoupledRoessler(AbstractDynamics):
    """CoupledRoessler neural mass model."""

    STATE_NAMES = ("x1", "y1", "z1", "x2", "y2", "z2")
    INITIAL_STATE = (1.0, -2.0, 0.0, 0.11, 0.2, 0.1)
    AUXILIARY_NAMES = ()

    DEFAULT_PARAMS = Bunch(
        a=0.2,
        b=0.2,
        c=5.7,
        k1=0.115,
        k2=0.0,
        omega1=0.18,
        omega2=0.22,
    )

    COUPLING_INPUTS = {}

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        a = params.a
        b = params.b
        c = params.c
        k1 = params.k1
        k2 = params.k2
        omega1 = params.omega1
        omega2 = params.omega2

        x1 = state[0]
        y1 = state[1]
        z1 = state[2]
        x2 = state[3]
        y2 = state[4]
        z2 = state[5]

        dx1_dt = (-omega1) * y1 - z1
        dy1_dt = omega1 * x1 + a * y1 + k1 * (y2 - y1)
        dz1_dt = b + z1 * (x1 - c)
        dx2_dt = (-omega2) * y2 - z2
        dy2_dt = omega2 * x2 + a * y2 + k2 * (y1 - y2)
        dz2_dt = b + z2 * (x2 - c)

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dx1_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dx1_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dy1_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dz1_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dx2_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dy2_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dz2_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.array([])

        return derivatives, auxiliaries
