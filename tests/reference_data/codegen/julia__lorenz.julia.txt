function Lorenz63!(dx, x, p, t = 0)

    (;sigma, rho, beta) = p

    X, Y, Z = x


    dx[1] = sigma .* (Y .- X)
    dx[2] = -Y .+ X .* (rho .- Z)
    dx[3] = X .* Y .- Z .* beta
    dx
end

# Parameter values
p = (sigma = 10.0, rho = 28.0, beta = 2.6666666666666665)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.0,
        10.0,
        0.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Lorenz63!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
