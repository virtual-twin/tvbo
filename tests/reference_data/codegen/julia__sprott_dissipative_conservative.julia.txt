function SprottDissipativeConservative!(dx, _x, p, t = 0)

    (;a, b, c) = p

    x, y, z = _x


    dx[1] = y .+ x .* z .+ a .* x .* y
    dx[2] = 1 .- 2 * x .^ 2 .+ b .* y .* z
    dx[3] = -x .^ 2 .- y .^ 2 .+ c .* x
    dx
end

# Parameter values
p = (a = 2.0, b = 1.0, c = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        1.0,
        0.0,
        0.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(SprottDissipativeConservative!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
