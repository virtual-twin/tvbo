import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    iSoma = _p.iSoma
    iDend = _p.iDend
    gLs = _p.gLs
    gLd = _p.gLd
    gNa = _p.gNa
    gKdr = _p.gKdr
    gCa = _p.gCa
    gKahp = _p.gKahp
    gKC = _p.gKC
    gc = _p.gc
    eNa = _p.eNa
    eCa = _p.eCa
    eK = _p.eK
    eL = _p.eL
    pp = _p.pp
    cm = _p.cm

    # State variables
    Vs = current_state[0]
    Vd = current_state[1]
    Cad = current_state[2]
    hs = current_state[3]
    ns = current_state[4]
    sd = current_state[5]
    cd = current_state[6]
    qd = current_state[7]
    Si = current_state[8]
    Wi = current_state[9]

    # Derived variables
    ICa_d = gCa * sd**2 * (Vd - eCa)
    alpha_cd = np.where(
        np.less(Vd, -10.0),
        0.684443445486297 * np.exp(0.0538720538720539 * Vd),
        0.275729683468338 * np.exp(-0.037037037037037 * Vd),
    )
    alpha_hs = 0.0117416385561416 * np.exp(-0.0555555555555556 * Vs)
    alpha_ms = np.where(
        np.equal(Vs, -46.9),
        0.08,
        (-15.008 - 0.32 * Vs) / (-1.0 + 8.08904403605717e-6 * np.exp(-0.25 * Vs)),
    )
    alpha_ns = np.where(
        np.equal(Vs, -24.9),
        0.0032,
        (-0.3984 - 0.016 * Vs) / (-1.0 + 0.00687406255749625 * np.exp(-0.2 * Vs)),
    )
    alpha_qd = functools.reduce(np.minimum, [0.01, 2.0e-5 * Cad])
    alpha_sd = 1.6 / (1.0 + 1.43332941456034 * np.exp(-0.072 * Vd))
    beta_hs = 4.0 / (1.0 + 0.0183156388887342 * np.exp(-0.2 * Vs))
    beta_ms = np.where(
        np.equal(Vs, -19.9),
        0.056,
        (5.572 + 0.28 * Vs) / (-1.0 + 53.5170342274912 * np.exp(0.2 * Vs)),
    )
    beta_ns = 0.0919698602928606 * np.exp(-0.025 * Vs)
    beta_sd = np.where(
        np.equal(Vd, -8.9),
        0.004,
        (0.178 + 0.02 * Vd) / (-1.0 + 5.92985641859115 * np.exp(0.2 * Vd)),
    )
    chi_d = functools.reduce(np.minimum, [1.0, 0.004 * Cad])
    beta_cd = np.where(
        np.less(Vd, -10.0),
        -alpha_cd + 0.275729683468338 * np.exp(-0.037037037037037 * Vd),
        0.0,
    )
    minf_s = alpha_ms / (alpha_ms + beta_ms)

    # Derivatives
    dVs_dt = (
        iSoma / pp
        - gLs * (Vs - eL)
        + gc * (Vd - Vs) / pp
        - gKdr * ns * (Vs - eK)
        - gNa * hs * minf_s**2 * (Vs - eNa)
    ) / cm
    dVd_dt = (
        -ICa_d
        + iDend / (1.0 - pp)
        - gLd * (Vd - eL)
        + gc * (Vs - Vd) / (1.0 - pp)
        - gKahp * qd * (Vd - eK)
        - cd * chi_d * gKC * (Vd - eK)
    ) / cm
    dCad_dt = -0.13 * ICa_d - 0.075 * Cad
    dhs_dt = alpha_hs - hs * (alpha_hs + beta_hs)
    dns_dt = alpha_ns - ns * (alpha_ns + beta_ns)
    dsd_dt = alpha_sd - sd * (alpha_sd + beta_sd)
    dcd_dt = alpha_cd - cd * (alpha_cd + beta_cd)
    dqd_dt = alpha_qd - qd * (0.001 + alpha_qd)
    dSi_dt = -0.00666666666666667 * Si
    dWi_dt = -0.5 * Wi

    derivatives = np.array(
        np.broadcast_arrays(
            dVs_dt,
            dVd_dt,
            dCad_dt,
            dhs_dt,
            dns_dt,
            dsd_dt,
            dcd_dt,
            dqd_dt,
            dSi_dt,
            dWi_dt,
        )
    )

    return derivatives
