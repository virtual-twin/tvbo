import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    c = _p.c

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = -y - z
    dy_dt = x + a * y
    dz_dt = b + z * (x - c)

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
