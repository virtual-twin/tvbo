import jax.numpy as jnp
import jax.scipy as jsp


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    c = _p.c

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = -(y**2) - 4 * y - 4 * z - a * x
    dy_dt = -(z**2) - 4 * x - 4 * z - a * y
    dz_dt = c - x**2 - 4 * x - 4 * y - a * z

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
