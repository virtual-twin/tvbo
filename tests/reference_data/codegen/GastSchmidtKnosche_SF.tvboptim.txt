import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class GastSchmidtKnosche_SF(AbstractDynamics):
    """GastSchmidtKnosche_SF neural mass model."""

    STATE_NAMES = ("A", "B", "V", "r")
    INITIAL_STATE = (0.1, 0.1, 0.1, 0.1)
    AUXILIARY_NAMES = ()

    DEFAULT_PARAMS = Bunch(
        Delta=2.0,
        I=0.0,
        J=21.2132,
        alpha=10.0,
        cr=1.0,
        cv=0.0,
        eta=1.0,
        tau=1.0,
        tau_A=10.0,
    )

    COUPLING_INPUTS = {
        "c_glob0": 1,
        "c_glob1": 1,
    }

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        Delta = params.Delta
        I = params.I
        J = params.J
        alpha = params.alpha
        cr = params.cr
        cv = params.cv
        eta = params.eta
        tau = params.tau
        tau_A = params.tau_A

        A = state[0]
        B = state[1]
        V = state[2]
        r = state[3]

        c_glob0 = (
            jnp.atleast_1d(coupling.c_glob0)[0] if hasattr(coupling, "c_glob0") else 0.0
        )
        c_glob1 = (
            jnp.atleast_1d(coupling.c_glob1)[0] if hasattr(coupling, "c_glob1") else 0.0
        )

        dA_dt = B / tau_A
        dB_dt = (-A - 2 * B + alpha * r) / tau_A
        dV_dt = (
            -A
            + I
            + J * r * tau
            + V**2
            + c_glob0 * cr
            + c_glob1 * cv
            + eta
            - jnp.pi**2 * r**2 * tau**2
        ) / tau
        dr_dt = (Delta / ((jnp.pi * tau)) + 2 * V * r) / tau

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dA_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dA_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dB_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dV_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dr_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.array([])

        return derivatives, auxiliaries
