function BetaTransformationMap!(dx, _x, p, t = 0)

    (;beta,) = p

    x = _x[1]


    dx[1] = (-floor(beta .* x) .+ beta .* x)
    dx
end

# Parameter values
p = (beta = 2.0,)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.25,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(BetaTransformationMap!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
