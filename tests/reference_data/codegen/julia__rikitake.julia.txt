




function Rikitake!(dx, _x, p, t = 0)

    (;mu, alpha) = p

    x, y, z = _x




    dx[1] = y .* z .- mu .* x
    dx[2] = x .* (z .- alpha) .- mu .* y
    dx[3] = 1 .- x .* y
    dx
end

# Parameter values
p = (mu = 1.0, alpha = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        1.0,
        0.0,
        0.6,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Rikitake!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
