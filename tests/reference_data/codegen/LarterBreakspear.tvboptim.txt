import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class LarterBreakspear(AbstractDynamics):
    """LarterBreakspear neural mass model."""

    STATE_NAMES = ("V", "W", "Z")
    INITIAL_STATE = (0.1, 0.1, 0.1)
    AUXILIARY_NAMES = ("Q_V", "Q_Z", "m_Ca", "m_K", "m_Na", "lc_0")

    DEFAULT_PARAMS = Bunch(
        C=0.1,
        I_ext=0.3,
        Q_Vmax=1.0,
        Q_Zmax=1.0,
        T_Ca=-0.01,
        T_K=0.0,
        T_Na=0.3,
        V_Ca=1.0,
        V_K=-0.7,
        V_L=-0.5,
        V_Na=0.53,
        V_T=0.0,
        Z_T=0.0,
        a_ee=0.4,
        a_ei=2.0,
        a_ie=2.0,
        a_ne=1.0,
        a_ni=0.4,
        b=0.1,
        delta_Ca=0.15,
        delta_K=0.3,
        delta_Na=0.15,
        delta_V=0.65,
        delta_Z=0.7,
        g_Ca=1.1,
        g_K=2.0,
        g_L=0.5,
        g_Na=6.7,
        phi=0.7,
        r_NMDA=0.25,
        t_scale=1.0,
        tau_K=1.0,
    )

    COUPLING_INPUTS = {
        "c_glob": 1,
        "local_coupling": 1,
    }

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        C = params.C
        I_ext = params.I_ext
        Q_Vmax = params.Q_Vmax
        Q_Zmax = params.Q_Zmax
        T_Ca = params.T_Ca
        T_K = params.T_K
        T_Na = params.T_Na
        V_Ca = params.V_Ca
        V_K = params.V_K
        V_L = params.V_L
        V_Na = params.V_Na
        V_T = params.V_T
        Z_T = params.Z_T
        a_ee = params.a_ee
        a_ei = params.a_ei
        a_ie = params.a_ie
        a_ne = params.a_ne
        a_ni = params.a_ni
        b = params.b
        delta_Ca = params.delta_Ca
        delta_K = params.delta_K
        delta_Na = params.delta_Na
        delta_V = params.delta_V
        delta_Z = params.delta_Z
        g_Ca = params.g_Ca
        g_K = params.g_K
        g_L = params.g_L
        g_Na = params.g_Na
        phi = params.phi
        r_NMDA = params.r_NMDA
        t_scale = params.t_scale
        tau_K = params.tau_K

        V = state[0]
        W = state[1]
        Z = state[2]

        c_glob = (
            jnp.atleast_1d(coupling.c_glob)[0] if hasattr(coupling, "c_glob") else 0.0
        )
        local_coupling = (
            jnp.atleast_1d(coupling.local_coupling)[0]
            if hasattr(coupling, "local_coupling")
            else 0.0
        )

        Q_V = 0.5 * Q_Vmax * (jnp.tanh((V - V_T) / delta_V) + 1)
        Q_Z = 0.5 * Q_Zmax * (jnp.tanh((Z - Z_T) / delta_Z) + 1)
        m_Ca = 0.5 * jnp.tanh((-T_Ca + V) / delta_Ca) + 0.5
        m_K = 0.5 * jnp.tanh((-T_K + V) / delta_K) + 0.5
        m_Na = 0.5 * jnp.tanh((-T_Na + V) / delta_Na) + 0.5
        lc_0 = Q_V * local_coupling

        dV_dt = t_scale * (
            I_ext * a_ne
            - Q_Z * Z * a_ie
            - W * g_K * (V - V_K)
            - g_L * (V - V_L)
            + m_Ca
            * (V - V_Ca)
            * (
                (-C) * a_ee * c_glob * r_NMDA
                - a_ee * r_NMDA * (1.0 - C) * (Q_V + lc_0)
                - g_Ca
            )
            - (V - V_Na)
            * (C * a_ee * c_glob + a_ee * (1.0 - C) * (Q_V + lc_0) + g_Na * m_Na)
        )
        dW_dt = phi * t_scale * (-W + m_K) / tau_K
        dZ_dt = b * t_scale * (I_ext * a_ni + Q_V * V * a_ei)

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dV_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dV_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dW_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dZ_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(Q_V), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(Q_Z), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(m_Ca), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(m_K), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(m_Na), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(lc_0), _per_node_shape),
            ],
            axis=0,
        )

        return derivatives, auxiliaries
