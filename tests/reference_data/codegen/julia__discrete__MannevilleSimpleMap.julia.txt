function MannevilleSimpleMap!(dx, _x, p, t = 0)

    (;epsilon,) = p

    x = _x[1]


    dx[1] = (-floor(x .* (1 .+ epsilon) .+ x .^ 2 .* (1 .- epsilon)) .+ x .* (1 .+ epsilon) .+ x .^ 2 .* (1 .- epsilon))
    dx
end

# Parameter values
p = (epsilon = 0.1,)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.4,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(MannevilleSimpleMap!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
