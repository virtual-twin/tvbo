import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    G1 = _p.G1
    G2 = _p.G2
    G3 = _p.G3
    G4 = _p.G4
    G5 = _p.G5
    G6 = _p.G6
    G7 = _p.G7
    G8 = _p.G8
    G9 = _p.G9
    G10 = _p.G10
    T1 = _p.T1
    T2 = _p.T2
    T3 = _p.T3
    T4 = _p.T4
    R = _p.R
    U = _p.U

    # Coupling
    c_glob = cX[0]

    # State variables
    x1 = current_state[0]
    x2 = current_state[1]
    x3 = current_state[2]
    x4 = current_state[3]
    x5 = current_state[4]
    x6 = current_state[5]
    x7 = current_state[6]
    x8 = current_state[7]

    # Functions
    def Sigm(v):
        return -1 / 2 + (1 + jnp.exp(-R * v)) ** (-1.0)

    # Derivatives
    dx1_dt = x2
    dx2_dt = (
        U + c_glob - 2 * x2 - G1 * Sigm(x1) - G2 * Sigm(x3) - G3 * Sigm(x5) - x1 / T1
    ) / T1
    dx3_dt = x4
    dx4_dt = (-2 * x4 + G8 * Sigm(x1) - G7 * Sigm(x3) - x3 / T2) / T2
    dx5_dt = x6
    dx6_dt = (-2 * x6 + G5 * Sigm(x1) + G6 * Sigm(x7) - G4 * Sigm(x5) - x5 / T3) / T3
    dx7_dt = x8
    dx8_dt = (-2 * x8 - G10 * Sigm(x7) - G9 * Sigm(x5) - x7 / T4) / T4

    derivatives = jnp.array(
        jnp.broadcast_arrays(
            dx1_dt, dx2_dt, dx3_dt, dx4_dt, dx5_dt, dx6_dt, dx7_dt, dx8_dt
        )
    )

    return derivatives
