import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class Gissinger(AbstractDynamics):
    """Gissinger neural mass model."""

    STATE_NAMES = ("Q", "D", "V")
    INITIAL_STATE = (3.0, 0.5, 1.5)
    AUXILIARY_NAMES = ()

    DEFAULT_PARAMS = Bunch(
        Gamma=0.9,
        mu=0.119,
        nu=0.1,
    )

    COUPLING_INPUTS = {}

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        Gamma = params.Gamma
        mu = params.mu
        nu = params.nu

        Q = state[0]
        D = state[1]
        V = state[2]

        dQ_dt = mu * Q - V * D
        dD_dt = (-nu) * D + V * Q
        dV_dt = Gamma - V + Q * D

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dQ_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dQ_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dD_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dV_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.array([])

        return derivatives, auxiliaries
