import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    mu = _p.mu
    alpha = _p.alpha

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = y * z - mu * x
    dy_dt = x * (z - alpha) - mu * y
    dz_dt = 1 - x * y

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
