import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    C = _p.C
    k = _p.k
    vr = _p.vr
    vt = _p.vt
    vpeak = _p.vpeak
    a = _p.a
    b = _p.b
    c = _p.c
    d = _p.d

    # State variables
    v = current_state[0]
    u = current_state[1]

    # Derivatives
    dv_dt = (-u + k * (v - vr) * (v - vt)) / C
    du_dt = a * (-u + b * (v - vr))

    derivatives = np.array(np.broadcast_arrays(dv_dt, du_dt))

    return derivatives
