import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    c = _p.c
    d = _p.d

    # State variables
    x = current_state[0]
    y = current_state[1]

    # Derived variables
    theta = c - d / (1 + x**2 + y**2)

    # Derivatives
    dx_dt = a + b * (x * np.cos(theta) - y * np.sin(theta))
    dy_dt = b * (x * np.sin(theta) + y * np.cos(theta))

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt))

    return derivatives
