import NaNMath
function Antidots!(dx, _x, p, t = 0)

    (;d0, c, B) = p

    x, y, vx, vy = _x


    xt = -0.5 .+ (-floor(1.0 * x) .+ 1.0 * x)
    yt = -0.5 .+ (-floor(1.0 * y) .+ 1.0 * y)
    rho = sqrt(xt .^ 2 .+ yt .^ 2)
    Ux = ifelse(rho < 0.5 * d0 .+ c, (-4 * (0.5 * d0 .+ c .- rho) .^ 3) .* xt ./ ((c .^ 4 .* (rho .+ 1.0e-12))), 0)
    Uy = ifelse(rho < 0.5 * d0 .+ c, (-4 * (0.5 * d0 .+ c .- rho) .^ 3) .* yt ./ ((c .^ 4 .* (rho .+ 1.0e-12))), 0)

    dx[1] = vx
    dx[2] = vy
    dx[3] = -Ux .+ 2 * B .* vy .* sqrt(2)
    dx[4] = -Uy .- 2 * B .* vx .* sqrt(2)
    dx
end

# Parameter values
p = (d0 = 0.5, c = 0.2, B = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.5,
        0.5,
        0.25,
        0.25,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Antidots!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
