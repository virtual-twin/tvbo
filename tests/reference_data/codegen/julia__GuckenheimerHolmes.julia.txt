function GuckenheimerHolmes!(dx, _x, p, t = 0)

    (;a, b, c, d, e, f) = p

    x, y, z = _x


    dx[1] = a .* x .+ x .* (y .^ 2 .+ z .^ 2) .+ d .* y .* z
    dx[2] = b .* y .+ y .* (x .^ 2 .+ z .^ 2) .+ e .* x .* z
    dx[3] = c .* z .+ z .* (x .^ 2 .+ y .^ 2) .+ f .* x .* y
    dx
end

# Parameter values
p = (a = 0.5, b = 0.5, c = 0.5, d = 1.0, e = 1.0, f = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.1,
        0.0,
        0.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(GuckenheimerHolmes!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
