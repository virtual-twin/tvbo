import NaNMath
function EpileptorCodim3!(dx, _x, p, t = 0)

    (;mu1_start, mu2_start, nu_start, mu1_stop, mu2_stop, nu_stop, b, R, c, dstar, Ks, modification, c_glob) = p

    x, y, z = _x


    E0 = mu2_start ./ sqrt(mu1_start .^ 2 .+ mu2_start .^ 2 .+ nu_start .^ 2)
    E1 = -mu1_start ./ sqrt(mu1_start .^ 2 .+ mu2_start .^ 2 .+ nu_start .^ 2)
    E2 = nu_start ./ sqrt(mu1_start .^ 2 .+ mu2_start .^ 2 .+ nu_start .^ 2)
    F0 = (mu1_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .+ nu_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) ./ sqrt((mu1_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .+ nu_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2 .+ (mu2_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .- nu_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop)) .^ 2 .+ (-mu1_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop) .- mu2_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2)
    F1 = (mu2_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .- nu_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop)) ./ sqrt((mu1_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .+ nu_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2 .+ (mu2_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .- nu_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop)) .^ 2 .+ (-mu1_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop) .- mu2_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2)
    F2 = (-mu1_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop) .- mu2_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) ./ sqrt((mu1_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .+ nu_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2 .+ (mu2_start .* (mu1_start .* mu2_stop .- mu1_stop .* mu2_start) .- nu_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop)) .^ 2 .+ (-mu1_start .* (mu1_stop .* nu_start .- mu1_start .* nu_stop) .- mu2_start .* (mu2_stop .* nu_start .- mu2_start .* nu_stop)) .^ 2)
    mu1 = -R .* (E1 .* cos(z) .+ F1 .* sin(z))
    mu2 = R .* (E0 .* cos(z) .+ F0 .* sin(z))
    nu = R .* (E2 .* cos(z) .+ F2 .* sin(z))
    cardano_a = mu1 / 2
    cardano_d = -mu2 .^ 3 / 27 .+ mu1 .^ 2 / 4
    xs = ifelse(cardano_d < 0, 2 * NaNMath.pow(cardano_a .^ 2 .- cardano_d, 1 // 6) .* cos(atan(NaNMath.sqrt(-cardano_d), cardano_a) / 3), ifelse(cardano_a .+ NaNMath.sqrt(cardano_d) >= 0, NaNMath.pow(cardano_a .+ NaNMath.sqrt(cardano_d), 1 // 3), NaNMath.pow(abs(cardano_a .+ NaNMath.sqrt(cardano_d)), 1 // 3) / 2) .+ ifelse(cardano_a .- NaNMath.sqrt(cardano_d) >= 0, NaNMath.pow(cardano_a .- NaNMath.sqrt(cardano_d), 1 // 3), NaNMath.pow(abs(cardano_a .- NaNMath.sqrt(cardano_d)), 1 // 3) / 2))

    dx[1] = -y
    dx[2] = x .^ 3 .- mu1 .- mu2 .* x .- y .* (nu .+ x .^ 2 .+ b .* x)
    dx[3] = -c .* (sqrt(y .^ 2 .+ (x .- xs) .^ 2) .- dstar .+ Ks .* c_glob .+ ifelse(modification > 0, (-1 // 2 .+ z) .^ 7 / 10, 0))
    dx
end

# Parameter values
p = (mu1_start = -0.02285, mu2_start = 0.3448, nu_start = 0.2014, mu1_stop = -0.07465, mu2_stop = 0.3351, nu_stop = 0.2053, b = 1.0, R = 0.4, c = 0.001, dstar = 0.3, Ks = 0.0, modification = 1.0, c_glob = 0.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.5,
        0.0,
        0.05,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(EpileptorCodim3!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
