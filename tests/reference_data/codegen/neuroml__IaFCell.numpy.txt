import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    tau = _p.tau
    leakReversal = _p.leakReversal
    thresh = _p.thresh
    reset = _p.reset

    # State variables
    v = current_state[0]

    # Derived variables
    I_ext = np.where(
        np.logical_and.reduce((np.greater_equal(t, 0.1), np.less(t, 0.2))), 0.3, 0.0
    )

    # Derivatives
    dv_dt = I_ext + (leakReversal - v) / tau

    derivatives = np.array(np.broadcast_arrays(dv_dt))

    return derivatives
