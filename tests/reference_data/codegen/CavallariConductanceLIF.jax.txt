import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    V_leak = _p.V_leak
    tau_m = _p.tau_m
    g_leak = _p.g_leak
    E_AMPA = _p.E_AMPA
    E_GABA = _p.E_GABA
    tau_r_AMPA = _p.tau_r_AMPA
    tau_d_AMPA = _p.tau_d_AMPA
    tau_r_GABA = _p.tau_r_GABA
    tau_d_GABA = _p.tau_d_GABA
    g_AMPA_rec = _p.g_AMPA_rec
    g_AMPA_tha = _p.g_AMPA_tha
    g_AMPA_cort = _p.g_AMPA_cort
    g_GABA = _p.g_GABA

    # Coupling
    S_AMPA_rec = cX[0]
    S_AMPA_tha = cX[1]
    S_AMPA_cort = cX[2]
    S_GABA = cX[3]

    # State variables
    v = current_state[0]
    x_AMPA = current_state[1]
    geff_AMPA = current_state[2]
    x_GABA = current_state[3]
    geff_GABA = current_state[4]
    t_ref = current_state[5]

    # Derived variables
    I_AMPA = geff_AMPA * (v - E_AMPA)
    I_GABA = geff_GABA * (v - E_GABA)
    integrating = jnp.where(jnp.greater(t_ref, 0), 0.0, 1.0)

    # Derivatives
    dv_dt = integrating * (V_leak - v - (I_AMPA + I_GABA) / g_leak) / tau_m
    dx_AMPA_dt = (
        -x_AMPA
        + tau_m
        * (
            S_AMPA_cort * g_AMPA_cort
            + S_AMPA_rec * g_AMPA_rec
            + S_AMPA_tha * g_AMPA_tha
        )
    ) / tau_r_AMPA
    dgeff_AMPA_dt = (x_AMPA - geff_AMPA) / tau_d_AMPA
    dx_GABA_dt = (-x_GABA + S_GABA * g_GABA * tau_m) / tau_r_GABA
    dgeff_GABA_dt = (x_GABA - geff_GABA) / tau_d_GABA
    dt_ref_dt = -1 + integrating

    derivatives = jnp.array(
        jnp.broadcast_arrays(
            dv_dt, dx_AMPA_dt, dgeff_AMPA_dt, dx_GABA_dt, dgeff_GABA_dt, dt_ref_dt
        )
    )

    return derivatives
