import NaNMath
function Epileptor!(dx, x, p, t = 0)

    (;a, b, c, d, r, s, x0, Iext, slope, Iext2, tau, aa, bb, Kvf, Kf, Ks, tt, modification, c_glob0, c_glob1, local_coupling) = p

    x1, y1, z, x2, y2, g = x


    Iext_total = Iext .+ local_coupling .* x1
    f1 = ifelse(x1 < 0, b .* x1 .- a .* x1 .^ 2, slope .- x2 .+ 0.6 * (-4.0 .+ z) .^ 2)
    f2 = ifelse(x2 < -0.25, 0, aa .* (0.25 .+ x2))
    h = ifelse(modification > 0, x0 .+ 3 ./ (1 .+ 0.00673794699908547 * exp(-10.0 * x1)), -4 * x0 .+ 4 * x1 .+ ifelse(z < 0, -0.1 * z .^ 7, 0))
    lfp = x2 .- x1

    dx[1] = tt .* (Iext_total .+ y1 .- z .+ Kvf .* c_glob0 .+ f1 .* x1)
    dx[2] = tt .* (c .- y1 .- d .* x1 .^ 2)
    dx[3] = r .* tt .* (h .- z .+ Ks .* c_glob0)
    dx[4] = tt .* (1.05 .+ Iext2 .+ x2 .- y2 .- x2 .^ 3 .- 0.3 * z .+ Kf .* c_glob1 .+ bb .* g)
    dx[5] = tt .* (f2 .- y2) ./ tau
    dx[6] = tt .* (0.001 * x1 .- 0.01 * g)
    dx
end

# Parameter values
p = (a = 1.0, b = 3.0, c = 1.0, d = 5.0, r = 0.00035, s = 4.0, x0 = -1.6, Iext = 3.1, slope = 0.0, Iext2 = 0.45, tau = 10.0, aa = 6.0, bb = 2.0, Kvf = 0.0, Kf = 0.0, Ks = 0.0, tt = 1.0, modification = 0.0, c_glob0 = 0.0, c_glob1 = 0.0, local_coupling = 0.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.1,
        0.1,
        3.0,
        -1.0,
        0.1,
        0.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Epileptor!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
