import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    omega = _p.omega
    f = _p.f
    d = _p.d
    beta = _p.beta

    # State variables
    x = current_state[0]
    v = current_state[1]

    # Derivatives
    dx_dt = v
    dv_dt = -(x**3) + f * jnp.cos(omega * t) - beta * x - d * v

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dv_dt))

    return derivatives
