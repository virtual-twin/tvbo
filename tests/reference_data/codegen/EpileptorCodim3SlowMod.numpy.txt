import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    mu1_Ain = _p.mu1_Ain
    mu2_Ain = _p.mu2_Ain
    nu_Ain = _p.nu_Ain
    mu1_Bin = _p.mu1_Bin
    mu2_Bin = _p.mu2_Bin
    nu_Bin = _p.nu_Bin
    mu1_Aend = _p.mu1_Aend
    mu2_Aend = _p.mu2_Aend
    nu_Aend = _p.nu_Aend
    mu1_Bend = _p.mu1_Bend
    mu2_Bend = _p.mu2_Bend
    nu_Bend = _p.nu_Bend
    b = _p.b
    R = _p.R
    c = _p.c
    cA = _p.cA
    cB = _p.cB
    dstar = _p.dstar
    Ks = _p.Ks
    modification = _p.modification
    G0 = mu2_Ain / np.sqrt(mu1_Ain**2 + mu2_Ain**2 + nu_Ain**2)
    G1 = -mu1_Ain / np.sqrt(mu1_Ain**2 + mu2_Ain**2 + nu_Ain**2)
    G2 = nu_Ain / np.sqrt(mu1_Ain**2 + mu2_Ain**2 + nu_Ain**2)
    H0 = (
        mu1_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
        + nu_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
    ) / np.sqrt(
        (
            mu1_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            + nu_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
        + (
            mu2_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            - nu_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
        )
        ** 2
        + (
            -mu1_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
            - mu2_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
    )
    H1 = (
        mu2_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
        - nu_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
    ) / np.sqrt(
        (
            mu1_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            + nu_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
        + (
            mu2_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            - nu_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
        )
        ** 2
        + (
            -mu1_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
            - mu2_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
    )
    H2 = (
        -mu1_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
        - mu2_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
    ) / np.sqrt(
        (
            mu1_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            + nu_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
        + (
            mu2_Ain * (mu1_Ain * mu2_Aend - mu1_Aend * mu2_Ain)
            - nu_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
        )
        ** 2
        + (
            -mu1_Ain * (mu1_Aend * nu_Ain - mu1_Ain * nu_Aend)
            - mu2_Ain * (mu2_Aend * nu_Ain - mu2_Ain * nu_Aend)
        )
        ** 2
    )
    L0 = mu2_Bin / np.sqrt(mu1_Bin**2 + mu2_Bin**2 + nu_Bin**2)
    L1 = -mu1_Bin / np.sqrt(mu1_Bin**2 + mu2_Bin**2 + nu_Bin**2)
    L2 = nu_Bin / np.sqrt(mu1_Bin**2 + mu2_Bin**2 + nu_Bin**2)
    M0 = (
        mu1_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
        + nu_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
    ) / np.sqrt(
        (
            mu1_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            + nu_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
        + (
            mu2_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            - nu_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
        )
        ** 2
        + (
            -mu1_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
            - mu2_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
    )
    M1 = (
        mu2_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
        - nu_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
    ) / np.sqrt(
        (
            mu1_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            + nu_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
        + (
            mu2_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            - nu_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
        )
        ** 2
        + (
            -mu1_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
            - mu2_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
    )
    M2 = (
        -mu1_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
        - mu2_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
    ) / np.sqrt(
        (
            mu1_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            + nu_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
        + (
            mu2_Bin * (mu1_Bin * mu2_Bend - mu1_Bend * mu2_Bin)
            - nu_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
        )
        ** 2
        + (
            -mu1_Bin * (mu1_Bend * nu_Bin - mu1_Bin * nu_Bend)
            - mu2_Bin * (mu2_Bend * nu_Bin - mu2_Bin * nu_Bend)
        )
        ** 2
    )

    # Coupling
    c_glob = cX[0]

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]
    uA = current_state[3]
    uB = current_state[4]

    # Derived variables
    A0 = R * (G0 * np.cos(uA) + H0 * np.sin(uA))
    A1 = R * (G1 * np.cos(uA) + H1 * np.sin(uA))
    A2 = R * (G2 * np.cos(uA) + H2 * np.sin(uA))
    B0 = R * (L0 * np.cos(uB) + M0 * np.sin(uB))
    B1 = R * (L1 * np.cos(uB) + M1 * np.sin(uB))
    B2 = R * (L2 * np.cos(uB) + M2 * np.sin(uB))
    C0 = A1 * B2 - A2 * B1
    C1 = A2 * B0 - A0 * B2
    C2 = A0 * B1 - A1 * B0
    normA = np.sqrt(A0**2 + A1**2 + A2**2)
    E0 = A0 / normA
    E1 = A1 / normA
    E2 = A2 / normA
    Fx0 = A2 * C1 - A1 * C2
    Fx1 = A0 * C2 - A2 * C0
    Fx2 = A1 * C0 - A0 * C1
    normF = np.sqrt(Fx0**2 + Fx1**2 + Fx2**2)
    F0 = Fx0 / normF
    F1 = Fx1 / normF
    F2 = Fx2 / normF
    mu1 = -R * (E1 * np.cos(z) + F1 * np.sin(z))
    mu2 = R * (E0 * np.cos(z) + F0 * np.sin(z))
    nu = R * (E2 * np.cos(z) + F2 * np.sin(z))
    disc = -1 / 27 * mu2**3 + (1 / 4) * mu1**2
    sqdisc = np.sqrt(abs(disc))
    xs = np.where(
        np.greater_equal(disc, 0),
        abs(sqdisc + (1 / 2) * mu1) ** (1 / 3)
        * np.cos((1 / 3) * np.arctan2(0, sqdisc + (1 / 2) * mu1))
        + abs((1 / 2) * mu1 - sqdisc) ** (1 / 3)
        * np.cos((1 / 3) * np.arctan2(0, (1 / 2) * mu1 - sqdisc)),
        (sqdisc**2 + (1 / 4) * mu1**2) ** (1 / 6)
        * np.cos((1 / 3) * np.arctan2(sqdisc, (1 / 2) * mu1))
        + (sqdisc**2 + (1 / 4) * mu1**2) ** (1 / 6)
        * np.cos((1 / 3) * np.arctan2(-sqdisc, (1 / 2) * mu1)),
    )

    # Derivatives
    dx_dt = -y
    dy_dt = x**3 - mu1 - mu2 * x - y * (nu + x**2 + b * x)
    dz_dt = -c * (
        np.sqrt(y**2 + (x - xs) ** 2)
        - dstar
        + Ks * c_glob
        + np.where(np.greater(modification, 0), (1 / 10) * (-1 / 2 + z) ** 7, 0)
    )
    duA_dt = cA
    duB_dt = cB

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dz_dt, duA_dt, duB_dt))

    return derivatives
