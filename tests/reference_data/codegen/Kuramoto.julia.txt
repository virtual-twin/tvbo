function Kuramoto!(dx, x, p, t = 0)

    (;omega, c_glob, local_coupling) = p

    theta = x[1]


    lc_0 = sin(local_coupling .* theta)
    I = c_glob .+ lc_0

    dx[1] = I .+ omega
    dx
end

# Parameter values
p = (omega = 1.0, c_glob = 0.0, local_coupling = 0.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.1,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Kuramoto!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
