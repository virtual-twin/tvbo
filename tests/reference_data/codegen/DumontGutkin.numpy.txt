import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    Delta_e = _p.Delta_e
    Delta_i = _p.Delta_i
    Gamma = _p.Gamma
    I_e = _p.I_e
    I_i = _p.I_i
    J_ee = _p.J_ee
    J_ei = _p.J_ei
    J_ie = _p.J_ie
    J_ii = _p.J_ii
    eta_e = _p.eta_e
    eta_i = _p.eta_i
    tau_e = _p.tau_e
    tau_i = _p.tau_i
    tau_s = _p.tau_s

    # Coupling
    c_glob = cX[0]

    # State variables
    V_e = current_state[0]
    V_i = current_state[1]
    r_e = current_state[2]
    r_i = current_state[3]
    s_ee = current_state[4]
    s_ei = current_state[5]
    s_ie = current_state[6]
    s_ii = current_state[7]

    # Derivatives
    dV_e_dt = (
        I_e
        + eta_e
        + V_e**2
        + s_ee * tau_e
        - s_ei * tau_e
        - np.pi**2 * r_e**2 * tau_e**2
    ) / tau_e
    dV_i_dt = (
        I_i
        + eta_i
        + V_i**2
        + s_ie * tau_i
        - s_ii * tau_i
        - np.pi**2 * r_i**2 * tau_i**2
    ) / tau_i
    dr_e_dt = (2 * V_e * r_e + Delta_e / (np.pi * tau_e)) / tau_e
    dr_i_dt = (2 * V_i * r_i + Delta_i / (np.pi * tau_i)) / tau_i
    ds_ee_dt = (c_glob - s_ee + J_ee * r_e) / tau_s
    ds_ei_dt = (-s_ei + J_ei * r_i) / tau_s
    ds_ie_dt = (-s_ie + Gamma * c_glob + J_ie * r_e) / tau_s
    ds_ii_dt = (-s_ii + J_ii * r_i) / tau_s

    derivatives = np.array(
        np.broadcast_arrays(
            dV_e_dt, dV_i_dt, dr_e_dt, dr_i_dt, ds_ee_dt, ds_ei_dt, ds_ie_dt, ds_ii_dt
        )
    )

    return derivatives
