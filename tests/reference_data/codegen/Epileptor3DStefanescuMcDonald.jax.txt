import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    N = _p.N
    x_0 = _p.x_0
    tau_z = _p.tau_z

    # Coupling
    c_glob = cX[0]

    # State variables
    x1 = current_state[0]
    z = current_state[1]

    # Derivatives
    dx1_dt = N * (c_glob + x1 - z - 1 / 3 * x1**3)
    dz_dt = (-z - 4 * x_0 + 4 * x1) / tau_z

    derivatives = jnp.array(jnp.broadcast_arrays(dx1_dt, dz_dt))

    return derivatives
