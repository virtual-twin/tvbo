import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    J0 = _p.J0

    # State variables
    theta = current_state[0]
    x = current_state[1]

    # Derivatives
    dtheta_dt = np.mod(
        theta + a * np.sin(2 * theta) - b * np.sin(4 * theta) - x * np.sin(theta),
        2 * np.pi,
    )
    dx_dt = -J0 * np.cos(theta)

    derivatives = np.array(np.broadcast_arrays(dtheta_dt, dx_dt))

    return derivatives
