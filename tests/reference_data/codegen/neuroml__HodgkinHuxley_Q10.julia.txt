




import NaNMath
function HodgkinHuxley_Q10!(dx, x, p, t = 0)

    (;C, g_Na, g_K, g_L, E_Na, E_K, E_L, I_amp, pulse_delay, pulse_duration, Q10) = p

    v, m, h, n = x



    I_ext = ifelse(pulse_delay <= t && t < pulse_delay .+ pulse_duration, I_amp, 0.0)
    alpha_h = 0.00271419454822054 * Q10 .* exp(-0.05 * v)
    alpha_m = Q10 .* ifelse(v == -40.0, 1.0, (4.0 .+ 0.1 * v) ./ (1.0 .- 0.0183156388887342 * exp(-0.1 * v)))
    alpha_n = Q10 .* ifelse(v == -55.0, 0.1, (0.55 .+ 0.01 * v) ./ (1.0 .- 0.00408677143846407 * exp(-0.1 * v)))
    beta_h = 1.0 * Q10 ./ (1.0 .+ 0.0301973834223185 * exp(-0.1 * v))
    beta_m = 0.108087223804836 * Q10 .* exp(-0.0555555555555556 * v)
    beta_n = 0.055468413760135 * Q10 .* exp(-0.0125 * v)

    dx[1] = (1000 * I_ext .- g_L .* (v .- E_L) .- g_K .* n .^ 4 .* (v .- E_K) .- g_Na .* h .* m .^ 3 .* (v .- E_Na)) ./ C
    dx[2] = alpha_m .* (1 .- m) .- beta_m .* m
    dx[3] = alpha_h .* (1 .- h) .- beta_h .* h
    dx[4] = alpha_n .* (1 .- n) .- beta_n .* n
    dx
end

# Parameter values
p = (C = 10.0, g_Na = 1200.0, g_K = 360.0, g_L = 3.0, E_Na = 50.0, E_K = -77.0, E_L = -54.3, I_amp = 0.08, pulse_delay = 100.0, pulse_duration = 100.0, Q10 = 0.333333)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        -65.0,
        0.052932,
        0.596121,
        0.317677,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(HodgkinHuxley_Q10!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
