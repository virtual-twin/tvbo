function CoupledStandardMaps3!(dx, x, p, t = 0)

    (;k0, k1, k2, Gamma) = p

    theta0, p0, theta1, p1, theta2, p2 = x


    dx[1] = (p0 .+ theta0 .+ k0 .* sin(theta0) .- Gamma .* (-sin(theta0 .- theta1) .- sin(theta0 .- theta2)) .- 2 * pi * floor((p0 .+ theta0 .+ k0 .* sin(theta0) .- Gamma .* (-sin(theta0 .- theta1) .- sin(theta0 .- theta2))) / (2 * pi)))
    dx[2] = (p0 .+ k0 .* sin(theta0) .- Gamma .* (-sin(theta0 .- theta1) .- sin(theta0 .- theta2)) .- 2 * pi * floor((p0 .+ k0 .* sin(theta0) .- Gamma .* (-sin(theta0 .- theta1) .- sin(theta0 .- theta2))) / (2 * pi)))
    dx[3] = (p1 .+ theta1 .+ k1 .* sin(theta1) .- Gamma .* (-sin(theta1 .- theta2) .+ sin(theta0 .- theta1)) .- 2 * pi * floor((p1 .+ theta1 .+ k1 .* sin(theta1) .- Gamma .* (-sin(theta1 .- theta2) .+ sin(theta0 .- theta1))) / (2 * pi)))
    dx[4] = (p1 .+ k1 .* sin(theta1) .- Gamma .* (-sin(theta1 .- theta2) .+ sin(theta0 .- theta1)) .- 2 * pi * floor((p1 .+ k1 .* sin(theta1) .- Gamma .* (-sin(theta1 .- theta2) .+ sin(theta0 .- theta1))) / (2 * pi)))
    dx[5] = (p2 .+ theta2 .+ k2 .* sin(theta2) .- Gamma .* (sin(theta0 .- theta2) .+ sin(theta1 .- theta2)) .- 2 * pi * floor((p2 .+ theta2 .+ k2 .* sin(theta2) .- Gamma .* (sin(theta0 .- theta2) .+ sin(theta1 .- theta2))) / (2 * pi)))
    dx[6] = (p2 .+ k2 .* sin(theta2) .- Gamma .* (sin(theta0 .- theta2) .+ sin(theta1 .- theta2)) .- 2 * pi * floor((p2 .+ k2 .* sin(theta2) .- Gamma .* (sin(theta0 .- theta2) .+ sin(theta1 .- theta2))) / (2 * pi)))
    dx
end

# Parameter values
p = (k0 = 1.0, k1 = 1.0, k2 = 1.0, Gamma = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.001,
        0.001,
        0.002,
        0.002,
        0.003,
        0.003,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(CoupledStandardMaps3!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
