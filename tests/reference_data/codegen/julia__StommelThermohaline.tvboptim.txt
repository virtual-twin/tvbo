import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class StommelThermohaline(AbstractDynamics):
    """StommelThermohaline neural mass model."""

    STATE_NAMES = ("T", "S")
    INITIAL_STATE = (0.3, 0.2)
    AUXILIARY_NAMES = ()

    DEFAULT_PARAMS = Bunch(
        eta1=3.0,
        eta2=1.0,
        eta3=0.3,
    )

    COUPLING_INPUTS = {}

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        eta1 = params.eta1
        eta2 = params.eta2
        eta3 = params.eta3

        T = state[0]
        S = state[1]

        dT_dt = eta1 - T - jnp.abs(T - S) * T
        dS_dt = eta2 - eta3 * S - jnp.abs(T - S) * S

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dT_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dT_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dS_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.array([])

        return derivatives, auxiliaries
