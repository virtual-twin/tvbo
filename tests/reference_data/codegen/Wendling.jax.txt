import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    A = _p.A
    B = _p.B
    G = _p.G
    a = _p.a
    b = _p.b
    g = _p.g
    C = _p.C
    v0 = _p.v0
    e0 = _p.e0
    r = _p.r
    p = _p.p
    C1 = C
    C2 = 0.8 * C
    C3 = 0.25 * C
    C4 = 0.25 * C
    C5 = 0.1 * C
    C6 = 0.1 * C
    C7 = 0.8 * C

    # Coupling
    c_glob = cX[0]

    # State variables
    y0 = current_state[0]
    y1 = current_state[1]
    y2 = current_state[2]
    y3 = current_state[3]
    y4 = current_state[4]
    y5 = current_state[5]
    y6 = current_state[6]
    y7 = current_state[7]
    y8 = current_state[8]
    y9 = current_state[9]

    # Functions
    def Sigm(v):
        return 2 * e0 / (1 + jnp.exp(r * (v0 - v)))

    # Derived variables
    lfp = y1 - y2 - y3

    # Derivatives
    dy0_dt = y5
    dy1_dt = y6
    dy2_dt = y7
    dy3_dt = y8
    dy4_dt = y9
    dy5_dt = -y0 * a**2 - 2 * a * y5 + A * a * Sigm(c_glob + y1 - y2 - y3)
    dy6_dt = -y1 * a**2 - 2 * a * y6 + A * a * (p + C2 * Sigm(C1 * y0))
    dy7_dt = -y2 * b**2 - 2 * b * y7 + B * C4 * b * Sigm(C3 * y0)
    dy8_dt = -y3 * g**2 - 2 * g * y8 + C7 * G * g * Sigm(C5 * y0 - C6 * y4)
    dy9_dt = -y4 * b**2 - 2 * b * y9 + B * b * Sigm(C3 * y0)

    derivatives = jnp.array(
        jnp.broadcast_arrays(
            dy0_dt,
            dy1_dt,
            dy2_dt,
            dy3_dt,
            dy4_dt,
            dy5_dt,
            dy6_dt,
            dy7_dt,
            dy8_dt,
            dy9_dt,
        )
    )

    return derivatives
