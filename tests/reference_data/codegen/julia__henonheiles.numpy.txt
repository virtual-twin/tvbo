import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # State variables
    x = current_state[0]
    y = current_state[1]
    px = current_state[2]
    py = current_state[3]

    # Derived variables
    H = -1 / 3 * y**3 + 0.5 * px**2 + 0.5 * py**2 + 0.5 * x**2 + 0.5 * y**2 + y * x**2

    # Derivatives
    dx_dt = px
    dy_dt = py
    dpx_dt = -x - 2 * x * y
    dpy_dt = y**2 - y - x**2

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dpx_dt, dpy_dt))

    return derivatives
