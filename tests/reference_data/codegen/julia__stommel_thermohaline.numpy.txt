import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    eta1 = _p.eta1
    eta2 = _p.eta2
    eta3 = _p.eta3

    # State variables
    T = current_state[0]
    S = current_state[1]

    # Derivatives
    dT_dt = eta1 - T - T * np.abs(T - S)
    dS_dt = eta2 - S * eta3 - S * np.abs(T - S)

    derivatives = np.array(np.broadcast_arrays(dT_dt, dS_dt))

    return derivatives
