




import NaNMath
function PinskyRinzelCA3!(dx, x, p, t = 0)

    (;iSoma, iDend, gLs, gLd, gNa, gKdr, gCa, gKahp, gKC, gc, eNa, eCa, eK, eL, pp, cm) = p

    Vs, Vd, Cad, hs, ns, sd, cd, qd, Si, Wi = x



    ICa_d = gCa .* sd .^ 2 .* (Vd .- eCa)
    alpha_cd = ifelse(Vd < -10.0, 0.684443445486297 * exp(0.0538720538720539 * Vd), 0.275729683468338 * exp(-0.037037037037037 * Vd))
    alpha_hs = 0.0117416385561416 * exp(-0.0555555555555556 * Vs)
    alpha_ms = ifelse(Vs == -46.9, 0.08, (-15.008 .- 0.32 * Vs) ./ (-1.0 .+ 8.08904403605717e-6 * exp(-0.25 * Vs)))
    alpha_ns = ifelse(Vs == -24.9, 0.0032, (-0.3984 .- 0.016 * Vs) ./ (-1.0 .+ 0.00687406255749625 * exp(-0.2 * Vs)))
    alpha_qd = (ifelse(2.0e-5 * Cad >= 0.01, 0.01, 2.0e-5 * Cad))
    alpha_sd = 1.6 ./ (1.0 .+ 1.43332941456034 * exp(-0.072 * Vd))
    beta_hs = 4.0 ./ (1.0 .+ 0.0183156388887342 * exp(-0.2 * Vs))
    beta_ms = ifelse(Vs == -19.9, 0.056, (5.572 .+ 0.28 * Vs) ./ (-1.0 .+ 53.5170342274912 * exp(0.2 * Vs)))
    beta_ns = 0.0919698602928606 * exp(-0.025 * Vs)
    beta_sd = ifelse(Vd == -8.9, 0.004, (0.178 .+ 0.02 * Vd) ./ (-1.0 .+ 5.92985641859115 * exp(0.2 * Vd)))
    chi_d = (ifelse(0.004 * Cad >= 1.0, 1.0, 0.004 * Cad))
    beta_cd = ifelse(Vd < -10.0, -alpha_cd .+ 0.275729683468338 * exp(-0.037037037037037 * Vd), 0.0)
    minf_s = alpha_ms ./ (alpha_ms .+ beta_ms)

    dx[1] = (iSoma ./ pp .- gLs .* (Vs .- eL) .+ gc .* (Vd .- Vs) ./ pp .- gKdr .* ns .* (Vs .- eK) .- gNa .* hs .* minf_s .^ 2 .* (Vs .- eNa)) ./ cm
    dx[2] = (-ICa_d .+ iDend ./ (1.0 .- pp) .- gLd .* (Vd .- eL) .+ gc .* (Vs .- Vd) ./ (1.0 .- pp) .- gKahp .* qd .* (Vd .- eK) .- cd .* chi_d .* gKC .* (Vd .- eK)) ./ cm
    dx[3] = -0.13 * ICa_d .- 0.075 * Cad
    dx[4] = alpha_hs .- hs .* (alpha_hs .+ beta_hs)
    dx[5] = alpha_ns .- ns .* (alpha_ns .+ beta_ns)
    dx[6] = alpha_sd .- sd .* (alpha_sd .+ beta_sd)
    dx[7] = alpha_cd .- cd .* (alpha_cd .+ beta_cd)
    dx[8] = alpha_qd .- qd .* (0.001 .+ alpha_qd)
    dx[9] = -0.00666666666666667 * Si
    dx[10] = -0.5 * Wi
    dx
end

# Parameter values
p = (iSoma = 0.75, iDend = 0.0, gLs = 0.1, gLd = 0.1, gNa = 30.0, gKdr = 15.0, gCa = 10.0, gKahp = 0.8, gKC = 15.0, gc = 2.1, eNa = 60.0, eCa = 80.0, eK = -75.0, eL = -60.0, pp = 0.5, cm = 3.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        -60.0,
        -60.0,
        0.0,
        0.999,
        0.001,
        0.009,
        0.007,
        0.0,
        0.0,
        0.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(PinskyRinzelCA3!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
