import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    R = _p.R

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = y - x
    dy_dt = -x * z
    dz_dt = -R + x * y

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
