import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    Iext = _p.Iext
    Ks = _p.Ks
    Kvf = _p.Kvf
    a = _p.a
    b = _p.b
    c = _p.c
    d = _p.d
    modification = _p.modification
    r = _p.r
    slope = _p.slope
    tt = _p.tt
    x0 = _p.x0

    # Coupling
    c_glob = cX[0]
    local_coupling = 0

    # State variables
    x1 = current_state[0]
    z = current_state[1]

    # Derived variables
    x1cond = np.where(
        np.less(x1, 0),
        a * x1**2 + x1 * (d - b),
        -slope - 0.6 * (-4.0 + z) ** 2 + d * x1,
    )
    zcond = np.where(np.less(z, 0), -0.1 * z**7, 0)
    h = np.where(
        np.greater(modification, 0),
        x0 + 3.0 / (1.0 + 0.00673794699908547 * np.exp(-10.0 * x1)),
        zcond - 4 * x0 + 4 * x1,
    )

    # Derivatives
    dx1_dt = tt * (Iext + c - z + Kvf * c_glob + local_coupling * x1 - x1 * x1cond)
    dz_dt = r * tt * (h - z + Ks * c_glob)

    derivatives = np.array(np.broadcast_arrays(dx1_dt, dz_dt))

    return derivatives
