import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    I = _p.I
    phi = _p.phi

    # State variables
    V = current_state[0]
    W = current_state[1]

    # Derivatives
    dV_dt = I + V - W - 1 / 3 * V**3
    dW_dt = phi * (V + a - W * b)

    derivatives = jnp.array(jnp.broadcast_arrays(dV_dt, dW_dt))

    return derivatives
