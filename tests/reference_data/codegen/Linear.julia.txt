function Linear!(dx, _x, p, t = 0)

    (;gamma, c_glob, local_coupling) = p

    x = _x[1]


    dx[1] = c_glob .+ gamma .* x .+ local_coupling .* x
    dx
end

# Parameter values
p = (gamma = -10.0, c_glob = 0.0, local_coupling = 0.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        0.1,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Linear!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
