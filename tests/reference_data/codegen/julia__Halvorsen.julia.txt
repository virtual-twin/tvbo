function Halvorsen!(dx, _x, p, t = 0)

    (;a, b, c) = p

    x, y, z = _x


    dx[1] = -y .^ 2 .- 4 * y .- 4 * z .- a .* x
    dx[2] = -z .^ 2 .- 4 * x .- 4 * z .- a .* y
    dx[3] = c .- x .^ 2 .- 4 * x .- 4 * y .- a .* z
    dx
end

# Parameter values
p = (a = 1.4, b = 0.4, c = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        -1.0,
        0.0,
        1.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Halvorsen!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
