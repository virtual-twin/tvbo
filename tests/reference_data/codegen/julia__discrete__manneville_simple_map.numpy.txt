import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # Parameters
    epsilon = _p.epsilon

    # State variables
    x = current_state[0]

    # Derivatives
    dx_dt = np.mod(x * (1 + epsilon) + x**2 * (1 - epsilon), 1)

    derivatives = np.array(np.broadcast_arrays(dx_dt))

    return derivatives
