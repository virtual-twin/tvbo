import numpy as np
import scipy.special


def dfun(current_state, t, cX, _p):
    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = np.sin(y)
    dy_dt = np.sin(z)
    dz_dt = np.sin(x)

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
