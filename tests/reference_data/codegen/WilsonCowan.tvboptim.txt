import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class WilsonCowan(AbstractDynamics):
    """WilsonCowan neural mass model."""

    STATE_NAMES = ("E", "I")
    INITIAL_STATE = (0.1, 0.1)
    AUXILIARY_NAMES = ("lc_0", "lc_1", "x_e", "x_i", "s_e", "s_i")

    DEFAULT_PARAMS = Bunch(
        P=0.0,
        Q=0.0,
        a_e=1.2,
        a_i=1.0,
        alpha_e=1.0,
        alpha_i=1.0,
        b_e=2.8,
        b_i=4.0,
        c_e=1.0,
        c_ee=12.0,
        c_ei=4.0,
        c_i=1.0,
        c_ie=13.0,
        c_ii=11.0,
        k_e=1.0,
        k_i=1.0,
        r_e=1.0,
        r_i=1.0,
        shift_sigmoid=1.0,
        tau_e=10.0,
        tau_i=10.0,
        theta_e=0.0,
        theta_i=0.0,
    )

    COUPLING_INPUTS = {
        "c_glob": 1,
        "local_coupling": 1,
    }

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        P = params.P
        Q = params.Q
        a_e = params.a_e
        a_i = params.a_i
        alpha_e = params.alpha_e
        alpha_i = params.alpha_i
        b_e = params.b_e
        b_i = params.b_i
        c_e = params.c_e
        c_ee = params.c_ee
        c_ei = params.c_ei
        c_i = params.c_i
        c_ie = params.c_ie
        c_ii = params.c_ii
        k_e = params.k_e
        k_i = params.k_i
        r_e = params.r_e
        r_i = params.r_i
        shift_sigmoid = params.shift_sigmoid
        tau_e = params.tau_e
        tau_i = params.tau_i
        theta_e = params.theta_e
        theta_i = params.theta_i

        E = state[0]
        I = state[1]

        c_glob = (
            jnp.atleast_1d(coupling.c_glob)[0] if hasattr(coupling, "c_glob") else 0.0
        )
        local_coupling = (
            jnp.atleast_1d(coupling.local_coupling)[0]
            if hasattr(coupling, "local_coupling")
            else 0.0
        )

        lc_0 = local_coupling * E
        lc_1 = I * local_coupling
        x_e = alpha_e * ((-I) * c_ei + P + c_ee * E + c_glob + lc_0 + lc_1 - theta_e)
        x_i = alpha_i * ((-I) * c_ii + Q + c_ie * E + lc_0 + lc_1 - theta_i)
        s_e = c_e / (1.0 + jnp.exp((-a_e) * (-b_e + x_e))) - 1.0 * shift_sigmoid / (
            jnp.exp(a_e * b_e) + 1.0
        )
        s_i = c_i / (1.0 + jnp.exp((-a_i) * (-b_i + x_i))) - 1.0 * shift_sigmoid / (
            jnp.exp(a_i * b_i) + 1.0
        )

        dE_dt = (-E + s_e * ((-E) * r_e + k_e)) / tau_e
        dI_dt = (-I + s_i * ((-I) * r_i + k_i)) / tau_i

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dE_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dE_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dI_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(lc_0), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(lc_1), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(x_e), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(x_i), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(s_e), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(s_i), _per_node_shape),
            ],
            axis=0,
        )

        return derivatives, auxiliaries
