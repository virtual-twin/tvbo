import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    z = _p.z

    # State variables
    x = current_state[0]

    # Derivatives
    dx_dt = np.where(
        np.greater(x, 0.5),
        -4 * x + 3,
        np.where(
            np.less(x, -0.5),
            -4 * x - 3,
            np.where(
                np.less_equal(np.abs(x), 0.5), x * (1 + np.abs(2 * x) ** (z - 1)), 0
            ),
        ),
    )

    derivatives = np.array(np.broadcast_arrays(dx_dt))

    return derivatives
