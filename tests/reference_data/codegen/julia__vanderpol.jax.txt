import jax.numpy as jnp
import jax.scipy as jsp


def dfun(current_state, t, cX, _p):
    # Parameters
    mu = _p.mu
    F = _p.F
    T = _p.T

    # State variables
    x = current_state[0]
    y = current_state[1]

    # Derivatives
    dx_dt = y
    dy_dt = -x + F * jnp.sin(2 * jnp.pi * t / T) + mu * y * (1 - x**2)

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dy_dt))

    return derivatives
