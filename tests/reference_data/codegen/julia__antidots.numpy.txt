import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    d0 = _p.d0
    c = _p.c
    B = _p.B

    # State variables
    x = current_state[0]
    y = current_state[1]
    vx = current_state[2]
    vy = current_state[3]

    # Derived variables
    xt = -0.5 + np.mod(1.0 * x, 1)
    yt = -0.5 + np.mod(1.0 * y, 1)
    rho = np.sqrt(xt**2 + yt**2)
    Ux = np.where(
        np.less(rho, c + 0.5 * d0),
        -4 * xt * (c - rho + 0.5 * d0) ** 3 / (c**4 * (1.0e-12 + rho)),
        0,
    )
    Uy = np.where(
        np.less(rho, c + 0.5 * d0),
        -4 * yt * (c - rho + 0.5 * d0) ** 3 / (c**4 * (1.0e-12 + rho)),
        0,
    )

    # Derivatives
    dx_dt = vx
    dy_dt = vy
    dvx_dt = -Ux + 2 * B * vy * np.sqrt(2)
    dvy_dt = -Uy - 2 * B * vx * np.sqrt(2)

    derivatives = np.array(np.broadcast_arrays(dx_dt, dy_dt, dvx_dt, dvy_dt))

    return derivatives
