function IntegrateAndFire!(dx, x, p, t = 0)

    (;leakReversal, tau, thresh, reset) = p

    v = x[1]


    dx[1] = (leakReversal .- v) ./ tau
    dx
end

# Parameter values
p = (leakReversal = -50.0, tau = 30.0, thresh = -55.0, reset = -70.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        -50.0,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(IntegrateAndFire!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
