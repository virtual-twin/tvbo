import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    eps = _p.eps

    # State variables
    x0 = current_state[0]
    x1 = current_state[1]
    x2 = current_state[2]
    x3 = current_state[3]

    # Derivatives
    dx0_dt = 2 - (eps * x3 + x0 * (1 - eps)) ** 2
    dx1_dt = 2 - (eps * x0 + x1 * (1 - eps)) ** 2
    dx2_dt = 2 - (eps * x1 + x2 * (1 - eps)) ** 2
    dx3_dt = 2 - (eps * x2 + x3 * (1 - eps)) ** 2

    derivatives = jnp.array(jnp.broadcast_arrays(dx0_dt, dx1_dt, dx2_dt, dx3_dt))

    return derivatives
