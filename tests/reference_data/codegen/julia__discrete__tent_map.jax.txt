import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    mu = _p.mu

    # State variables
    x = current_state[0]

    # Derivatives
    dx_dt = jnp.where(jnp.less(x, 0.5), mu * x, mu * (1 - x))

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt))

    return derivatives
