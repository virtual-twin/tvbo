import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    eps = _p.eps
    nu = _p.nu
    r = _p.r

    # State variables
    x = current_state[0]
    y = current_state[1]

    # Derived variables
    mu = (1 - jnp.exp(-r)) / r

    # Derivatives
    dx_dt = jnp.mod(x + nu * (1 + mu * y) + eps * mu * nu * jnp.cos(2 * jnp.pi * x), 1)
    dy_dt = (y + eps * jnp.cos(2 * jnp.pi * x)) * jnp.exp(-r)

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dy_dt))

    return derivatives
