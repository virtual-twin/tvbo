import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class Generic2dOscillator(AbstractDynamics):
    """Generic2dOscillator neural mass model."""

    STATE_NAMES = ("V", "W")
    INITIAL_STATE = (0.1, 0.1)
    AUXILIARY_NAMES = ()

    DEFAULT_PARAMS = Bunch(
        I=0.0,
        a=-2.0,
        alpha=1.0,
        b=-10.0,
        beta=1.0,
        c=0.0,
        d=0.02,
        e=3.0,
        f=1.0,
        g=0.0,
        gamma=1.0,
        tau=1.0,
    )

    COUPLING_INPUTS = {
        "c_glob": 1,
        "local_coupling": 1,
    }

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        I = params.I
        a = params.a
        alpha = params.alpha
        b = params.b
        beta = params.beta
        c = params.c
        d = params.d
        e = params.e
        f = params.f
        g = params.g
        gamma = params.gamma
        tau = params.tau

        V = state[0]
        W = state[1]

        c_glob = (
            jnp.atleast_1d(coupling.c_glob)[0] if hasattr(coupling, "c_glob") else 0.0
        )
        local_coupling = (
            jnp.atleast_1d(coupling.local_coupling)[0]
            if hasattr(coupling, "local_coupling")
            else 0.0
        )

        dV_dt = (
            d
            * tau
            * (
                I * gamma
                - V**3 * f
                + V**2 * e
                + V * g
                + V * local_coupling
                + W * alpha
                + c_glob * gamma
            )
        )
        dW_dt = d * (V**2 * c + V * b - W * beta + a) / tau

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dV_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dV_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dW_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.array([])

        return derivatives, auxiliaries
