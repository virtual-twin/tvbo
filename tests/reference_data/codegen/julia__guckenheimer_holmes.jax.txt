import jax.numpy as jnp
import jax.scipy as jsp


def dfun(current_state, t, cX, _p):
    # Parameters
    a = _p.a
    b = _p.b
    c = _p.c
    d = _p.d
    e = _p.e
    f = _p.f

    # State variables
    x = current_state[0]
    y = current_state[1]
    z = current_state[2]

    # Derivatives
    dx_dt = a * x + x * (y**2 + z**2) + d * y * z
    dy_dt = b * y + y * (x**2 + z**2) + e * x * z
    dz_dt = c * z + z * (x**2 + y**2) + f * x * y

    derivatives = jnp.array(jnp.broadcast_arrays(dx_dt, dy_dt, dz_dt))

    return derivatives
