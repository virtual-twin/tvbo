import numpy as np


def dfun(current_state, t, cX, _p):
    # Parameters
    omega = _p.omega
    f = _p.f
    d = _p.d

    # State variables
    x = current_state[0]
    v = current_state[1]

    # Derivatives
    dx_dt = v
    dv_dt = -np.sin(x) + f * np.cos(omega * t) - d * v

    derivatives = np.array(np.broadcast_arrays(dx_dt, dv_dt))

    return derivatives
