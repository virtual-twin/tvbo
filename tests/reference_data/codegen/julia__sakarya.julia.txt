




function Sakarya!(dx, _x, p, t = 0)

    (;a, b, m) = p

    x, y, z = _x




    dx[1] = y .+ a .* x .+ y .* z
    dx[2] = y .* z .- x .* z
    dx[3] = b .- z .- m .* x .* y
    dx
end

# Parameter values
p = (a = 1.0, b = 1.0, m = 1.0)


using OrdinaryDiffEqTsit5

# Initial conditions (flat state vector; multi-mode SVs are length-n_modes blocks)
u0 = [
        -2.8976045,
        3.8877978,
        3.07465,
    ]

# Define time span
tspan = (0.0, 1000) # Adjust time span as needed

prob = ODEProblem(Sakarya!, u0, tspan, p)


# Solve
sol = solve(prob, Tsit5(); saveat=0.01)
