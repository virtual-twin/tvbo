import jax
import jax.numpy as jnp
import jax.scipy as jsp
from typing import Tuple

from tvboptim.experimental.network_dynamics.core.bunch import Bunch
from tvboptim.experimental.network_dynamics.dynamics.base import AbstractDynamics


class EpileptorCodim3(AbstractDynamics):
    """EpileptorCodim3 neural mass model."""

    STATE_NAMES = ("x", "y", "z")
    INITIAL_STATE = (0.5, 0.0, 0.05)
    AUXILIARY_NAMES = (
        "E0",
        "E1",
        "E2",
        "F0",
        "F1",
        "F2",
        "mu1",
        "mu2",
        "nu",
        "cardano_a",
        "cardano_d",
        "xs",
    )

    DEFAULT_PARAMS = Bunch(
        Ks=0.0,
        R=0.4,
        b=1.0,
        c=0.001,
        dstar=0.3,
        modification=1.0,
        mu1_start=-0.02285,
        mu1_stop=-0.07465,
        mu2_start=0.3448,
        mu2_stop=0.3351,
        nu_start=0.2014,
        nu_stop=0.2053,
    )

    COUPLING_INPUTS = {
        "c_glob": 1,
    }

    def dynamics(
        self,
        t: float,
        state: jnp.ndarray,
        params: Bunch,
        coupling: Bunch,
        external: Bunch,
    ) -> Tuple[jnp.ndarray, jnp.ndarray]:
        Ks = params.Ks
        R = params.R
        b = params.b
        c = params.c
        dstar = params.dstar
        modification = params.modification
        mu1_start = params.mu1_start
        mu1_stop = params.mu1_stop
        mu2_start = params.mu2_start
        mu2_stop = params.mu2_stop
        nu_start = params.nu_start
        nu_stop = params.nu_stop

        x = state[0]
        y = state[1]
        z = state[2]

        c_glob = (
            jnp.atleast_1d(coupling.c_glob)[0] if hasattr(coupling, "c_glob") else 0.0
        )

        E0 = mu2_start / jnp.sqrt(mu1_start**2 + mu2_start**2 + nu_start**2)
        E1 = (-mu1_start) / jnp.sqrt(mu1_start**2 + mu2_start**2 + nu_start**2)
        E2 = nu_start / jnp.sqrt(mu1_start**2 + mu2_start**2 + nu_start**2)
        F0 = (
            mu1_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
            + nu_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
        ) / jnp.sqrt(
            (
                mu1_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                + nu_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                (-mu1_start) * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
                - mu2_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                mu2_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                - nu_start * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
            )
            ** 2
        )
        F1 = (
            mu2_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
            - nu_start * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
        ) / jnp.sqrt(
            (
                mu1_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                + nu_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                (-mu1_start) * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
                - mu2_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                mu2_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                - nu_start * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
            )
            ** 2
        )
        F2 = (
            (-mu1_start) * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
            - mu2_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
        ) / jnp.sqrt(
            (
                mu1_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                + nu_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                (-mu1_start) * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
                - mu2_start * ((-mu2_start) * nu_stop + mu2_stop * nu_start)
            )
            ** 2
            + (
                mu2_start * (mu1_start * mu2_stop - mu1_stop * mu2_start)
                - nu_start * ((-mu1_start) * nu_stop + mu1_stop * nu_start)
            )
            ** 2
        )
        mu1 = (-R) * (E1 * jnp.cos(z) + F1 * jnp.sin(z))
        mu2 = R * (E0 * jnp.cos(z) + F0 * jnp.sin(z))
        nu = R * (E2 * jnp.cos(z) + F2 * jnp.sin(z))
        cardano_a = mu1 / 2
        cardano_d = mu1**2 / 4 - mu2**3 / 27
        xs = jnp.where(
            jnp.less(cardano_d, 0),
            2
            * (cardano_a**2 - cardano_d) ** (1 / 6)
            * jnp.cos(jnp.arctan2(jnp.sqrt(-cardano_d), cardano_a) / 3),
            jnp.where(
                jnp.greater_equal(cardano_a + jnp.sqrt(cardano_d), 0),
                (cardano_a + jnp.sqrt(cardano_d)) ** (1 / 3),
                abs(cardano_a + jnp.sqrt(cardano_d)) ** (1 / 3) / 2,
            )
            + jnp.where(
                jnp.greater_equal(cardano_a - jnp.sqrt(cardano_d), 0),
                (cardano_a - jnp.sqrt(cardano_d)) ** (1 / 3),
                abs(cardano_a - jnp.sqrt(cardano_d)) ** (1 / 3) / 2,
            ),
        )

        dx_dt = -y
        dy_dt = x**3 - mu2 * x - mu1 - y * (nu + b * x + x**2)
        dz_dt = (-c) * (
            jnp.sqrt((x - xs) ** 2 + y**2)
            - dstar
            + jnp.where(jnp.greater(modification, 0), (z - 1 / 2) ** 7 / 10, 0)
            + Ks * c_glob
        )

        # Determine per-node shape from the first state variable; broadcast
        # any scalar derivatives or auxiliaries so jnp.stack sees uniform
        # rank-1 arrays.
        _per_node_shape = jnp.atleast_1d(dx_dt).shape

        derivatives = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(dx_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dy_dt), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(dz_dt), _per_node_shape),
            ],
            axis=0,
        )

        auxiliaries = jnp.stack(
            [
                jnp.broadcast_to(jnp.atleast_1d(E0), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(E1), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(E2), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(F0), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(F1), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(F2), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(mu1), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(mu2), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(nu), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(cardano_a), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(cardano_d), _per_node_shape),
                jnp.broadcast_to(jnp.atleast_1d(xs), _per_node_shape),
            ],
            axis=0,
        )

        return derivatives, auxiliaries
