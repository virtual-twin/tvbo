import jax.numpy as jnp


def dfun(current_state, t, cX, _p):
    # Parameters
    Iext2 = _p.Iext2
    Iext = _p.Iext
    Kf = _p.Kf
    Ks = _p.Ks
    Kvf = _p.Kvf
    a = _p.a
    aa = _p.aa
    b = _p.b
    bb = _p.bb
    c = _p.c
    d = _p.d
    modification = _p.modification
    r = _p.r
    slope = _p.slope
    tau = _p.tau
    tt = _p.tt
    x0 = _p.x0

    # Coupling
    c_glob0 = cX[0]
    c_glob1 = cX[1]
    local_coupling = 0

    # State variables
    g = current_state[0]
    x1 = current_state[1]
    x2 = current_state[2]
    y1 = current_state[3]
    y2 = current_state[4]
    z = current_state[5]

    # Derived variables
    x1cond = jnp.where(
        jnp.less(x1, 0), b * x1 - a * x1**2, slope - x2 + 0.6 * (-4.0 + z) ** 2
    )
    y2cond = jnp.where(jnp.less(x2, -0.25), 0.0, aa * (0.25 + x2))
    zcond = jnp.where(jnp.less(z, 0), -0.1 * z**7, 0.0)
    h = jnp.where(
        jnp.greater(modification, 0),
        x0 + 3 / (1 + 0.00673794699908547 * jnp.exp(-10.0 * x1)),
        zcond - 4 * x0 + 4 * x1,
    )

    # Derivatives
    dg_dt = tt * (0.001 * x1 - 0.01 * g)
    dx1_dt = tt * (Iext + y1 - z + Kvf * c_glob0 + local_coupling * x1 + x1 * x1cond)
    dx2_dt = tt * (1.05 + Iext2 + x2 - y2 - x2**3 - 0.3 * z + Kf * c_glob1 + bb * g)
    dy1_dt = tt * (c - y1 - d * x1**2)
    dy2_dt = tt * (y2cond - y2) / tau
    dz_dt = r * tt * (h - z + Ks * c_glob0)

    derivatives = jnp.array(
        jnp.broadcast_arrays(dg_dt, dx1_dt, dx2_dt, dy1_dt, dy2_dt, dz_dt)
    )

    return derivatives
